const express = require("express"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json({ limit: "15mb" })); const ANTHROPIC_API_KEY = process.env.ANTHROPIC_API_KEY; app.get("/health", (req, res) => { res.json({ status: "ok" }); }); app.post("/describe", async (req, res) => { try { const { image, mediaType, question } = req.body; const cleanQuestion = typeof question === "string" ? question.trim() : ""; if (!image && !cleanQuestion) { return res.status(400).json({ error: "Please ask a question or provide a photo." }); } let content = []; if (image) { content.push({ type: "image", source: { type: "base64", media_type: mediaType || "image/jpeg", data: image } }); } let prompt; if (image && cleanQuestion) { prompt = ` You are Know N'Go, an accessible personal AI concierge. The user provided a photo and asked: "${cleanQuestion}" Answer the user's question about the photo directly and clearly. Return ONLY valid JSON in this exact format: { "confident": "Your direct answer", "hazard": "Any important safety concern, or Nothing hazardous spotted", "needs_closer_photo": "What requires a closer photo, or Nothing — the photo is clear", "suggestion": "One useful next action" } `; } else if (image) { prompt = ` You are Know N'Go, an accessible personal AI concierge. Describe the provided photo clearly for someone who may not be able to see it. Return ONLY valid JSON in this exact format: { "confident": "Clearly describe what you can see", "hazard": "Any important safety concern, or Nothing hazardous spotted", "needs_closer_photo": "What requires a closer photo, or Nothing — the photo is clear", "suggestion": "One useful next action" } `; } else { prompt = ` You are Know N'Go, a helpful personal AI concierge. Answer this user's question directly: "${cleanQuestion}" Be clear, practical, and concise. `; } content.push({ type: "text", text: prompt }); const response = await fetch( "https://api.anthropic.com/v1/messages", { method: "POST", headers: { "Content-Type": "application/json", "x-api-key": ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" }, body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 1000, messages: [ { role: "user", content } ] }) } ); const data = await response.json(); if (!response.ok) { console.error("Anthropic error:", data); return res.status(502).json({ error: "Know N'Go could not get an answer right now." }); } const textBlock = (data.content || []).find( item => item.type === "text" ); const text = textBlock && textBlock.text ? textBlock.text.trim() : ""; if (!image) { return res.json({ answer: text || "No answer was returned." }); } try { const cleaned = text .replace(/```json/gi, "") .replace(/```/g, "") .trim(); const parsed = JSON.parse(cleaned); return res.json(parsed); } catch { return res.json({ confident: text || "No description was returned.", hazard: "", needs_closer_photo: "", suggestion: "" }); } } catch (error) { console.error("Server error:", error); res.status(500).json({ error: "Something went wrong on the server." }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Know N'Go server running on port ${PORT}`); });
