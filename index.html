const express = require('express'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(express.json({ limit: '15mb' })); const ANTHROPIC_API_KEY = process.env.ANTHROPIC_API_KEY; function textPrompt(question) { return `You are Know N'Go, a helpful personal AI concierge. Answer the user's question clearly, directly, and conversationally. User question: "${question}" Respond with ONLY valid JSON in this format: {"answer":"your response here"}`; } function photoPrompt(question) { return `You are Know N'Go, a visual assistant. Look carefully at the photo and respond ONLY with valid JSON using these keys: "confident": what you can clearly see and the answer to the user's question, "hazard": any possible danger, or "Nothing hazardous spotted", "needs_closer_photo": anything you cannot determine confidently, or "Nothing — the photo is clear", "suggestion": one useful next action. ${question ? `The user specifically asked: "${question}". Prioritize answering that question.` : ''} Respond with ONLY the JSON object.`; } app.post('/describe', async (req, res) => { try { const { image, mediaType, question } = req.body; if (!image && !question) { return res.status(400).json({ error: 'Please enter a question or provide a photo.' }); } let content; if (image) { content = [ { type: 'image', source: { type: 'base64', media_type: mediaType || 'image/jpeg', data: image } }, { type: 'text', text: photoPrompt(question) } ]; } else { content = [ { type: 'text', text: textPrompt(question) } ]; } const response = await fetch( 'https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' }, body: JSON.stringify({ model: 'claude-sonnet-4-6', max_tokens: 1000, messages: [ { role: 'user', content: content } ] }) } ); const data = await response.json(); if (!response.ok) { console.error('Anthropic API error:', data); return res.status(502).json({ error: 'The AI service had a problem. Please try again.' }); } const textBlock = (data.content || []).find( item => item.type === 'text' ); const raw = textBlock?.text || ''; try { const clean = raw .replace(/```json/g, '') .replace(/```/g, '') .trim(); const parsed = JSON.parse(clean); res.json(parsed); } catch (error) { res.json({ answer: raw || 'No response was returned.' }); } } catch (error) { console.error('Server error:', error); res.status(500).json({ error: 'Something went wrong. Please try again.' }); } }); app.get('/health', (req, res) => { res.send('ok'); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log( `Know N'Go server running on port ${PORT}` ); });
